\section{Introduzione}

\subsection{Di cosa si occupa il Machine Learning?}
Si occupa di risolvere problemi complessi tramite tecniche algoritmiche specifiche. L'approccio generale del ML è
\begin{itemize}
    \item definire una classe di modelli del problema su insieme di \textbf{parametri} $\Theta$
    \item definire una metrica di valutazione, ovvero una misura dell'\textbf{errore} dei modelli
    \item modificare i parametri $\Theta$ per minimizzare l'errore sui \textbf{dati di training}
\end{itemize}
Quindi possiamo dire che \texttt{il machine learning è un procedimento di ottimizzazione}.

Il ML tratta di problemi di \textbf{ottimizzazione}!
\begin{itemize}
    \item Il \texttt{training set} fa sì che il tuning dei parametri è dato dalla esperienza passata, quindi una sorta di \textbf{apprendimento}.
    \item La \texttt{soluzione} spesso non esiste una soluzione in forma chiusa
\end{itemize}
Utilizziamo \textbf{tecniche iterative} per approssimare iterativamente la soluzione.

\subsection{Features}
Ogni informazione realativa a un dato che ne descrive una sua proprietà è feature (caratteristica).
\begin{itemize}
    \item Le features sono gli input del processo di apprendimento.
    \item L'apprendimento è molto sensibile alla scelta delle features.
\end{itemize}
Determinare delle buone caratteristiche è complesso.
\begin{itemize}
    \item \textbf{Approccio tradizionale}: determinare durante il preprocessing delle buone features, e applicare un metodo semplice ma robusto di apprendimento, ad esempio qualche tecnica lineare.
    \item \textbf{Approccio ``deep''}: fornire in input dati grezzi e delegare alla macchina il compito di sintetizzare nuove features (rappresentatione interna).
\end{itemize}
\texttt{Deep learning} è implementato attraverso reti neurali profonde.\\
La rete neurale è profonda quando si hanno layers multipli:
\imgfig[0.55]{2026-09-24_11-59-11.png}
Ogni layer sintetizza nuove features in funzione delle precedenti.

    \item \textbf{Machine-Learning tradizionale}: prendete un esperto, chiedetegli di quali features avrebbe bisogno per risolvere un dato problema, raccogliete un insieme di osservazioni e lasciate alla macchina il compito di apprendere l'associazione

\subsection{Alberi di decisione}
Definiamo \textbf{traning set} un insieme di \texttt{esempi di allenamento}
\begin{align*}
    \langle x^{(i)}, y^{(i)}\rangle
\end{align*}
con
\begin{itemize}
    \item $x^{(i)} \in X$ insieme degli inputs
    \item $y^{(i)} \in Y$ insieme delgi outputs, \textbf{groudn truth}
    \item $i$ indice dell'istanza di training
\end{itemize}
Definiamo \textbf{problema} trovare la funzione che ``mappa'' $x^{(i)}$ a $y^{(i)}$
\begin{itemize}
    \item $Y$ discreto: problema di classificazione (predizione di una classe)
    \item $Y$ continuo: problema di regressione (predizione di un valore)
\end{itemize}
Le tecniche di Machine Learning richiedono la scelta preliminare di uno \texttt{spazio di funzioni} $H$, all'interno del quale cercare la soluzione che \texttt{meglio approssima} i dati di training. \\
\underline{Un modello è un modo di specificare e calcolare una funzione nello}\\
\underline{spazio delle ipotesi $H$.}

\paragraph{Expressività del modello}
Definiamo $X_1 \times X_2 \times \dots \times X_n$ con $X_i = \{\text{true, false}\}$ \\
Possiamo esprimere $Y=X_2 \lor X_5$?

\subsubsection{Costruzione Top-down induttiva}
Main loop:
\begin{enumerate}
    \item assegnare al nodo corrente il ``miglior'' attributo $X_i$
    \item creare un nodo figlio per ogni possibile valore di $X_i$ e propagare i dati verso i figli a seconda del loro valore
    \item per ogni nodo figlio, se tutti i dati del training set associati al nodo hanno una stessa etichetta $y$, marcare il nodo come foglia con etichetta $y$, altrimenti ripetere dal primo punto
\end{enumerate}
Ma quale è il ``miglior'' attributo?
\figura{\alberiAB}
Definiamo \textbf{entropia} di una variabile aleatoria $X$
\begin{align*}
    H(X)=- \sum_{i=1}^{n} P(X=i)\log_2 P(X=i)
\end{align*}
dove $n$ è il numero dei possibili valori di $n$.

\figura{%

\subsubsection{Teoria della Informazione (Shannon)}
L'Entropia è la quantità media di informazione prodotta da una sorgente stocastica di dati.
\begin{itemize}
    \item Un evento con entropia 1 non trasmette Informazioni
    \begin{align*}
        I(1)=0
    \end{align*}
    \item Dati due evento indipendenti con probabilità $p_1, p_2$ la probabilità congiunta è $p_1p_2$ ma l'informazione acquisita è la somma delle informazioni dei due eventi indipendenti
    \begin{align*}
        I(p_1p_2)= I(p_1)+I(p_2)
    \end{align*}
    \item Ci aspettiamo che l'informazione sia antimonotona rispetto alla probabilità ed è quindi naturale definire
    \begin{align*}
        I(p)=-\log(p)
    \end{align*}
\end{itemize}

\subsubsection*{Teoria dei codici (Shannon)}
L'Entropia misura il numero medio di bits richiesti per trasmettere il valore prodotto da una sorgente stocastica $X$. \\
\underline{Se gli eventi non hanno la stessa probabilità possiamo migliorare la codifica!!}

\paragraph{Entropia di $X$}
\begin{align*}
    H(X) = -\sum_{i=1}^{n} P(X=i)\log_2 P(X=i)
\end{align*}

\paragraph{Entropia Condizionale di $X$ dato uno specifico $Y=v$}
\begin{align*}
    H(X|Y=v) = -\sum_{i=1}^{n} P(X=i|Y=v)\log_2 P(X=i|Y=v)
\end{align*}

\paragraph{Entropia Condizionale di $X$ dato $Y$}
(media ponderata sugli $m$ possibili valori di $Y$)
\begin{align*}
    H(X|Y) = \sum_{v=1}^{m} P(Y=v)H(X|Y=v)
\end{align*}

\paragraph{Guadagno Informativo tra $X$ e $Y$}
\begin{align*}
    I(X,Y) = H(X)-H(X|Y) = H(Y)-H(Y|X)
\end{align*}

\subsubsection{Meglio A o B?}
Misuriamo la riduzione di entropia della variabile target $Y$ conseguente alla conoscenza di un qualche attributo $X$, cioè il guadagno informativo $I(Y,X)$ tra $Y$ e $X$.
\figura{\alberiAB}
\begin{align*}
    H(Y)       &= -(29/64)\cdot\log_2(29/64)-(35/64)\cdot\log_2(35/64) = .994\\
    H(Y|A=T)   &= -(21/26)\cdot\log_2(21/26)-(5/26)\cdot\log_2(5/26)   = .706\\
    H(Y|A=F)   &= -(8/38)\cdot\log_2(8/38)-(30/38)\cdot\log_2(30/38)   = .742\\
    H(Y|A)     &= .706\cdot 26/64 + .742\cdot 38/64 = .726\\
    I(Y,A)     &= H(Y)-H(Y|A) = .994-.726 = .288
\end{align*}
Per $B$ otteniamo $H(Y|B) = .872$ e $I(Y,B) = .122$
\begin{center}
    \textcolor{red}{Quindi, A è migliore!}
\end{center}

\subsection{Overfitting}
Consideriamo l'errore relativo all'ipotesi $h \in H$
\begin{itemize}
    \item sul training set: $\text{error}_{train}(h)$
    \item sul data set (globale): $\text{error}_{\mathcal{D}}(h)$
\end{itemize}
Definiamo \texttt{overfitting} quando esiste un'ipotesi $h' \in H$ tale che
\begin{align*}
    \text{error}_{train}(h') < \text{error}_{train}(h) \quad\text{ma}\quad
    \text{error}_{\mathcal{D}}(h') > \text{error}_{\mathcal{D}}(h)
\end{align*}
\imgfig[0.8]{img/overfitt.png}
Il problema? Non conosciamo $\mathcal{D}$! Quindi semplicement eprendiamo i dati a noi disposizione e li dividiamo in:
\begin{itemize}
    \item   training set: per costruire l'ipotesi $h$
    \item   test set: utilizzato per misurare accuracy e overfitting $h$
\end{itemize}
Tipicamente si riserva una ulteriore porzione, il \textbf{test set}, per la valutazione finale.
\subsubsection{Evitare overfitting sugli alberi di decisione}
\begin{itemize}
    \item \texttt{early stopping}: terminare la costruzione dell'albero non appena il miglioramento del modello non è più statisticamente significativo. Ad esempio, quando il guadagno informativo è inferiore ad una certa soglia, o il numero dei dati relativo al nodo è troppo piccolo.
    \item \texttt{post-pruning}: si sviluppa l'intero albero e quindi si procede a potarlo all'indietro.
    \begin{itemize}
        \item \underline{post-pruning finalizzato alla riduzione dell'errore}: costruire un albero di decisione completo per il training set, quindi ripetere la seguente operazione finché ogni ulteriore pruning non migliora l'accuratezza della predizione:
        \begin{enumerate}
            \item per ogni sottoalbero, valutare (sul validation set) l'impatto della sua rimozione sulla accuratezza della classificazione
            \item effettuare (in modo greedy) il pruning del sottoalbero che ottimizza l'accuracy
        \end{enumerate}
    \end{itemize}
\end{itemize}
\subsubsection{Gini's impurity}
Gini's impurity (coefficiente di Gini) misura la probabilità che un generico elemento sia mal classificato in base alla classificazione corrente (è una possibile alternativa alla nozione di entropia).\\
Date $m$ categorie, sia $f_i$ la frazione dei dati con etichetta $i$. Questa è pari alla probabilità che un qualche input appartenga alla categoria $i$. La probabilità che venga classificato in modo errato è dunque $1-f_i$, e il coefficiente di Gini è semplicemente la media pesata di questa quantità su tutte le possibili categorie, cioè
\begin{align*}
    I_G(f) & = \sum_{i=1}^{m} f_i(1-f_i) \\
    & = \sum_{i=1}^{m}\left(f_i - f_i^2\right) \\
    & = \sum_{i=1}^{m} f_i - \sum_{i=1}^{m} f_i^2 \\
    & = \boxed{1 - \sum_{i=1}^{m} f_i^2}
\end{align*}
La metrica è applicata su ogni nodo figlio, e i valori sono sommati in modo pesato, in modo del tutto analogo a quanto avviene per l'entropia, al fine di ottenere una misura della qualità dell'attributo.
\subsubsection*{Gini: Meglio A o B?}
Valutiamo gli split utilizzando il coefficiente di Gini.
\figura{\alberiAB}
\begin{align*}
    I_G(A=T) &= 1-(21/26)^2-(5/26)^2 = .310\\
    I_G(A=F) &= 1-(8/38)^2-(30/38)^2 = .332\\
    I_G(A)   &= .310\cdot 26/64 + .332\cdot 38/64 = .323
\end{align*}
\begin{align*}
    I_G(B=T) &= 1-(18/51)^2-(33/51)^2 = .456\\
    I_G(B=F) &= 1-(11/13)^2-(2/13)^2 = .260\\
    I_G(B)   &= .456\cdot 51/64 + .260\cdot 13/64 = .416
\end{align*}

\subsection{Approccio probabilistico}
Siamo interessati ad approssimare funzioni. Invece di calcolare
\begin{align*}
    f : X \to Y
\end{align*}
possiamo calcolare
\begin{align*}
    p : X \to P(Y|X)
\end{align*}


\subsubsection{Variabile aleatoria}
Una \textbf{variabile aleatoria} $X$ denota l'esito di un fenomeno riguardo al quale sussiste incertezza, come ad esempio il risultato di un processo stocastico. Ad esempio
\begin{itemize}
    \item $X = \text{true}$ se uno studente scelto a caso in aula è biondo
    \item $X = $ nome dello studente
    \item $X = \text{true}$ se due studenti scelti a caso sono nati lo stesso giorno dell'anno
\end{itemize}
Definiamo $P(X)$ come la frazione di volte in cui $X$ è vero, in esecuzioni ripetute dell'esperimento.

\subsubsection{Spazio campionario ed eventi}
\begin{itemize}
    \item l'insieme $\Omega$ dei possibili esiti di un esperimento casuale è detto \textbf{spazio campionario} (sample space)
    \item una variabile aleatoria è una funzione misurabile su $\Omega$: discreta (es. $\Omega \to \{m,f\}$) o continua (es. $\Omega \to \R$)
    \item un \textbf{evento} è un qualche sottoinsieme di $\Omega$; siamo interessati alla probabilità di eventi specifici, anche condizionati da altri eventi
\end{itemize}
La scelta dello spazio campionario richiede cautela: spesso l'intuizione presuppone (erroneamente) che gli eventi elementari abbiano tutti la stessa probabilità.\\
\textbf{Esempio}: nel lancio di due dadi il risultato è un numero tra 2 e 12, ma questi valori non hanno tutti la stessa probabilità; conviene invece considerare come spazio campionario tutte le coppie dei valori dei due dadi, e definire $X$ come la somma dei due risultati.
\figura{%
  \begin{tikzpicture}
    \draw[fill=cyan!25] (0,0) rectangle (5,3);
    \draw[fill=red!45] (3.3,1.5) circle (1.1);
    \node[font=\footnotesize] at (1.2,2.7) {Sample space};
    \node[font=\footnotesize] at (3.35,1.55) {Event A};
  \end{tikzpicture}%
}
La probabilità $P(A)$ dell'evento $A$ è il rapporto tra l'area di $A$ e l'area dell'intero spazio campionario (supposto uniforme).

\subsubsection{Assiomi della teoria della probabilità}
\begin{align*}
    0 &\le P(A) \le 1\\
    P(\text{True}) &= 1 \qquad P(\text{False}) = 0\\
    P(A \lor B) &= P(A) + P(B) - P(A \land B)
\end{align*}

\subsubsection{Teoremi derivati}
\textit{Dimostrazioni omesse.}
\begin{align*}
\boxed{P(\lnot A) = 1-P(A)}\end{align*}
\begin{align*}
\boxed{P(A) = P(A \land B) + P(A \land \lnot B)}\end{align*}

\subsubsection{Variabili aleatorie multivalore}
$A$ è una variabile aleatoria discreta a $k$ valori se può assumere esattamente uno dei valori $\{\nu_1,\nu_2,\dots,\nu_k\}$.
\begin{align*}
    P(A=\nu_i \land A=\nu_j) &= 0 \quad \text{se } i\neq j\\
    P(A=\nu_1 \lor A=\nu_2 \lor \dots \lor A=\nu_k) &= 1
\end{align*}

\subsubsection{Probabilità condizionata}
La probabilità condizionata dell'evento $A$ dato l'evento $B$ è definita come
\begin{align*}
    P(A|B) = \frac{P(A\land B)}{P(B)}
\end{align*}
\figura{%
  \begin{tikzpicture}
    \draw[fill=cyan!25] (0,0) rectangle (6,3.2);
    \draw[fill=yellow!70] (2,1.5) circle (1.1);
    \draw[fill=red!45, fill opacity=0.65] (3,1.5) circle (1.1);
    \node[font=\footnotesize] at (1.2,3) {Sample space};
    \node[font=\small] at (1.4,1.5) {B};
    \node[font=\small] at (3.6,1.5) {A};
  \end{tikzpicture}%
}
\textbf{Corollario (chain rule)}:
\begin{align*}
    P(A\land B) = P(B)\cdot P(A|B) = P(A)\cdot P(B|A)
\end{align*}

\subsubsection{Eventi indipendenti}
Due eventi $A$ e $B$ sono detti \textbf{indipendenti} se
\begin{align*}
    P(A|B) = P(A)
\end{align*}
ovvero l'evento $B$ non ha influenza sull'evento $A$. Come corollario
\begin{align*}
    P(A\land B) = P(A)\cdot P(B)
\end{align*}
e, siccome $P(A\land B) = P(B|A)\cdot P(A)$, abbiamo anche $P(B|A)=P(B)$: cioè anche $B$ è indipendente da $A$.

\subsubsection{Regola di Bayes}
\begin{align*}
    P(A|B) = \frac{P(A)\cdot P(B|A)}{P(B)}
\end{align*}


\subsubsection{Posterior, likelihood, prior e marginale}
In generale, per ogni $i,j$
\begin{align*}
    P(Y=y_i|X=x_j) = \frac{P(Y=y_i)\cdot P(X=x_j|Y=y_i)}{\sum_i P(Y=y_i)\cdot P(X=x_j|Y=y_i)}
\end{align*}
dove il denominatore si ottiene marginalizzando (sommando su $Y$) $P(X=x_j) = \sum_i P(X=x_j,Y=y_i)$. In notazione compatta
\begin{align*}
    \underbrace{P(Y|X)}_{\text{posterior}} = \frac{\overbrace{P(X|Y)}^{\text{likelihood}}\cdot\overbrace{P(Y)}^{\text{prior}}}{\underbrace{P(X)}_{\text{marginal likelihood}}}
\end{align*}
``Marginale'' in quanto stiamo marginalizzando (sommando/integrando) su $Y$.

\subsubsection{Uso della distribuzione congiunta}
Disponendo della distribuzione congiunta possiamo stimare la probabilità di \textbf{qualunque evento} esprimibile come combinazione logica delle features.
\begin{align*}
    P(E) = \sum_{\text{row} \in E} P(\text{row})
\end{align*}
\textbf{Esempio}: calcoliamo $P(M, \text{poor})$ dalla tabella precedente
\begin{align*}
    P(M, \text{poor}) &= P(M, \text{poor}, \le 40) + P(M, \text{poor}, > 40)\\
    &= 0.33 + 0.13 = 0.46
\end{align*}

\subsubsection{Inferenza con la distribuzione congiunta}
È altrettanto semplice calcolare la probabilità condizionata di un evento $E_1$ dato un altro evento $E_2$
\begin{align*}
    P(E_1|E_2) = \frac{P(E_1 \land E_2)}{P(E_2)} = \frac{\sum_{\text{row}\in E_1\land E_2} P(\text{row})}{\sum_{\text{row}\in E_2} P(\text{row})}
\end{align*}
Riprendiamo l'esempio precedente e calcoliamo $P(\text{M}|\text{poor})$
\begin{align*}
    P(\text{M}|\text{poor}) = \frac{P(\text{M}, \text{poor})}{P(\text{poor})} = \frac{0.46}{0.75} = 0.61
\end{align*}

\subsubsection*{Indipendenza condizionale}
Due eventi $X_i$ e $X_j$ sono \textbf{indipendenti dato} $Y$ se
\begin{align*}
    P(X_i|X_j, Y) = P(X_i|Y)
\end{align*}

\subsubsection*{Naïve Bayes}
Per la \textbf{chain rule}
\begin{align*}
    P(X_1,X_2|Y) = P(X_1|X_2,Y)\cdot P(X_2|Y)
\end{align*}
e per l'\textbf{indipendenza condizionale}
\begin{align*}
    P(X_1,X_2|Y) = P(X_1|Y)\cdot P(X_2|Y)
\end{align*}
In generale
\begin{align*}
    P(X_1,X_2,\dots,X_n|Y) = \prod_{i} P(X_i|Y)
\end{align*}
Quanti parametri per $P(X_1,\dots,X_n|Y)$ (nel caso booleano)?
\begin{itemize}
    \item senza indipendenza condizionale
    \item per Naïve Bayes
\end{itemize}

\subsubsection{Naïve Bayes: riassunto}
\textbf{Bayes rule}
\begin{align*}
    P(Y=y_i|X_1,\dots,X_n) = \frac{P(Y=y_i)\cdot P(X_1,\dots,X_n|Y=y_i)}{P(X_1,\dots,X_n)}
\end{align*}
\textbf{Naïve Bayes}
\begin{align*}
    P(Y=y_i|X_1,\dots,X_n) = \frac{P(Y=y_i)\cdot \prod_j P(X_j|Y=y_i)}{P(X_1,\dots,X_n)}
\end{align*}
Classificazione di un nuovo dato $x_{\text{new}} = \langle x_1,\dots,x_n\rangle$
\begin{align*}
    Y_{\text{new}} = \argmax_{y_i}\ P(Y=y_i)\cdot \prod_j P(X_j=x_j|Y=y_i)
\end{align*}
\subsubsection{Tecnica di apprendimento: stima dei parametri (MLE’s)}
Variabili aleatorie discrete $X_i$, $Y$.\\
\textbf{Training} (\textit{maximum likelihood estimates}): per tutti i valori $y_k$ di $Y$, stimare
\begin{align*}
    \pi_k = P(Y=y_k) = \frac{\#_D\{Y=y_k\}}{|D|}
\end{align*}
per ogni possibile valore $x_{ij}$ di $X_i$, stimare
\begin{align*}
    \theta_{ijk} = P(X_i = x_{ij} \mid Y = y_k) = \frac{\#_D\{X_i=x_{ij} \land Y=y_k\}}{\#_D\{Y=y_k\}}
\end{align*}
dove $\#_D\{\cdot\}$ indica il numero di istanze del dataset $D$ che soddisfano la condizione, e $|D|$ è la dimensione del dataset.

\textbf{Classificazione} di $a_{\text{new}} = \langle a_1,\dots,a_n\rangle$
\begin{align*}
    Y_{\text{new}} &= \argmax_{y_k}\ P(Y=y_k)\cdot \prod_i P(X_i=a_i|Y=y_k)\\
    &= \argmax_{k}\ \pi_k \cdot \prod_i \theta_{ijk}
\end{align*}
supponendo che $a_i = x_{ij}$, cioè che $a_i$ sia il $j$-esimo tra i possibili valori discreti dell'attributo $X_i$.

\subsection{Maximum Likelihood Estimation (MLE)}
\begin{gather*}
    \pi_k = P(Y=y_k) = \frac{\#\mathcal{D}\{Y=y_k\}}{|\mathcal{D}|}\\[1ex]
    \theta_{ijk} = P(X_i=x_{ij} \mid Y=y_k) = \frac{\#\mathcal{D}\{X_i=x_{ij} \land Y=y_k\}}{\#\mathcal{D}\{Y=y_k\}}
\end{gather*}
\paragraph{Perchè?} Supponiamo di lanciare una moneta (truccata) 10 volte, otteniamo 6 testa ($H$) e 4 code ($T$), possiamo naturalmente concludere che
\begin{align*}
    P(H) = .6 \\
    P(T)=.4
\end{align*}
Supponiamo ora di avere due possibilità:
\begin{enumerate}
    \item la moneta è reale, quindi $P(H)=.5$ e $P(T)=.5$
    \item la moneta è false, quindi $P(H)=.7$ e $P(T)=.3$
\end{enumerate}
\textit{Basandoci sulle nostre osservazioni, quale è il nostro caso?}\\
\vspace{1px}\\
La distribuzione di \textbf{Bernoulli} abbiamo 2 possibili output 0 o 1, con probabilità $\theta$ e $1-\theta$.\\
Sia $X^n$ il numero di 0 in una sequenza di $n$ lanci: $X^n$ segue una \hl{distribuzione binomiale}
\begin{align*}
    P(X^n=\alpha_0 \mid \theta) = \binom{n}{\alpha_0}\cdot \theta^{\alpha_0}\cdot (1-\theta)^{\alpha_1}
\end{align*}
dove $\alpha_1 = n-\alpha_0$ è il numero di 1 nella sequenza (ovvero $\alpha_0+\alpha_1=n$).\\
\vspace{1px}\\
\paragraph{Continiamo l'esempio di prima} $n=10$, $\alpha_0=6$ e $\alpha_1=4$.\\
Se $\theta=.5$ e $1-\theta=.5$
\begin{align*}
    P(X^n=6 \mid \theta) = \binom{n}{\alpha_0}\cdot \theta^{\alpha_0}\cdot (1-\theta)^{\alpha_1} = \binom{10}{6}\cdot \left(\frac{1}{2}\right)^{6}\cdot \left(\frac{1}{2}\right)^{4} = .205
\end{align*}
Se $\theta=.7$ e $1-\theta=.3$
\begin{align*}
    P(X^n=6 \mid \theta) = \binom{n}{\alpha_0}\cdot \theta^{\alpha_0}\cdot (1-\theta)^{\alpha_1} = \binom{10}{6}\cdot \left(\frac{7}{10}\right)^{6}\cdot \left(\frac{3}{10}\right)^{4} = .200
\end{align*}
Quindi $\theta=.5$ è leggermente più verosimile di $\theta=.7$.
\begin{center}
    \textcolor{red}{Qual è il valore più verosimile per $\theta$?}
\end{center}
\paragraph{Calcoliamo MLE}
\begin{align*}
    \hat{\theta} &= \arg\max_{\theta}\; P(X^n=\alpha_0 \mid \theta) =\\
                 &= \arg\max_{\theta}\; \theta^{\alpha_0}\cdot (1-\theta)^{\alpha_1}
\end{align*}
In modo equivalente possiamo cercare il $\theta$ che massimizza il logaritmo dell'espressione precedente
\begin{align*}
    \ln\left(\theta^{\alpha_0}\cdot (1-\theta)^{\alpha_1}\right) = \alpha_0\ln(\theta) + \alpha_1\ln(1-\theta)
\end{align*}
Derivando rispetto a $\theta$ otteniamo
\begin{align*}
    \frac{\alpha_0}{\theta} - \frac{\alpha_1}{1-\theta} = \frac{\alpha_0 - \alpha_0\theta - \alpha_1\theta}{\theta\cdot(1-\theta)}
\end{align*}
che si annulla per
\begin{align*}
    \theta = \frac{\alpha_0}{\alpha_0+\alpha_1} = \frac{\alpha_0}{n}
\end{align*}
\paragraph{Caso mulivariabile} \textbf{Distribuzione discreta} (es. lancio di un dado): $k$ possibili esiti $\{1,2,\dots,k\}$ con probabilità $\theta_i$ dove $\sum_i \theta_i = 1$.\\
Sequenza di $n$ lanci: \hl{distribuzione multinomiale}
\begin{align*}
    P(X^n=\alpha_i \mid \theta) = c_{\alpha_i}\prod_i \theta_i^{\alpha_i}
\end{align*}
dove $\alpha_i$ è il numero di $i$ nella sequenza e $c_{\overline{\alpha_i}}$ è una costante combinatoria che non dipende da $\theta$.
\begin{align*}
    \textbf{MLE}: \quad \theta_i = \frac{\alpha_i}{\sum_i \alpha_i} = \frac{\alpha_i}{n}
\end{align*}
